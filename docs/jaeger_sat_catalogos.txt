-- interface genérica para el repositorio
IRepositoryGeneric {
    Version:string
    Title:string
    Lastupdate:date
    Builder:string
    Load();
    Save();
    Import(DataTable);
}

-- interface para el repositorio
IRepositoryContext<T> : IReposityGeneric {
    FileName:string
    Revision:string
    WorkingFolder:string
    Items:List<T>
    T Search(string query);
    Restore();
    ZIP();
}

-- implementación de interfaces
RepositoryContext : IRepositoryContext<T> : IReposityGeneric  where T : class, new() {

}

Repositorio : RepositoryContext<TObject> {
}

Repositorio.Load();
Repositorio.Load(string fileName);
Repositorio.AddWorkingFolder(string path);
Repositorio.Save().ZIP();
